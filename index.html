// Vervang de functie haalAdresOp door deze vriendelijkere versie: function haalAdresOp(type) { const spinner = document.getElementById(type === 'start' ? 'gpsStartSpinner' : 'gpsStopSpinner'); const targetInput = document.getElementById(type === 'start' ? 'afhaalAdres' : 'brengAdres'); if (!("geolocation" in navigator)) { console.warn("Geolocatie wordt niet ondersteund."); return; } spinner.classList.remove('d-none'); navigator.geolocation.getCurrentPosition( async (position) => { const lat = position.coords.latitude; const lon = position.coords.longitude; try { const response = await fetch(`https://nominatim.openstreetmap.org/reverse?format=json&lat=${lat}&lon=${lon}`); const data = await response.json(); if (data && data.address) { const a = data.address; const straat = a.road || a.pedestrian || a.suburb || ''; const huisnr = a.house_number || ''; const stad = a.city || a.town || a.village || ''; let geformeerdAdres = `${straat} ${huisnr}`.trim(); if (stad) geformeerdAdres += `, ${stad}`; targetInput.value = geformeerdAdres || data.display_name; } else { targetInput.value = `GPS: ${lat.toFixed(5)}, ${lon.toFixed(5)}`; } } catch (e) { console.error("Adres ophalen mislukt:", e); targetInput.value = `GPS: ${lat.toFixed(5)}, ${lon.toFixed(5)}`; } finally { spinner.classList.add('d-none'); } }, (error) => { spinner.classList.add('d-none'); console.warn("GPS niet beschikbaar of geweigerd:", error.message); // Geen alert pop-up meer, de gebruiker kan nu gewoon handmatig het adres invullen. }, { enableHighAccuracy: true, timeout: 8000 } ); }
